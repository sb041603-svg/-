# ============================== 공통 함수 ==============================
def norm(v) -> str:
    if v is None:
        return ""
    s = str(v).strip()
    s = re.sub(r"\[[^\]]*\]", "", s)
    s = re.sub(r"\([^)]*\)", "", s)
    s = re.sub(r"\s+", "", s)
    return s.replace("-", "").replace("_", "").lower()


def read_csv(path: Path, **kwargs) -> pd.DataFrame:
    return pd.read_csv(path, dtype=str, keep_default_na=False, na_filter=False, encoding="utf-8-sig", **kwargs)


def find_gtfs_base(src: str, work: Path) -> Path:
    srcp = Path(src)
    if srcp.is_dir():
        base = srcp
    else:
        extract = work / "outer"
        extract.mkdir(parents=True, exist_ok=True)
        with zipfile.ZipFile(srcp) as z:
            z.extractall(extract)
        base = extract
        # 외부 ZIP 안에 KTDB 내부 ZIP이 있는 구조를 지원
        inner_zips = list(extract.rglob("*.zip"))
        if inner_zips:
            inner = work / "inner"
            inner.mkdir(parents=True, exist_ok=True)
            with zipfile.ZipFile(inner_zips[0]) as z:
                z.extractall(inner)
            base = inner
    routes = list(base.rglob("routes.txt"))
    if not routes:
        raise FileNotFoundError(f"routes.txt를 찾지 못했습니다: {src}")
    return routes[0].parent


def require_columns(df, required, name):
    missing = [c for c in required if c not in df.columns]
    if missing:
        raise ValueError(f"{name}에 필요한 컬럼이 없습니다: {missing}; 실제 컬럼={df.columns.tolist()}")


def service_dates(calendar: pd.DataFrame, calendar_dates: pd.DataFrame | None, year: int):
    """calendar와 calendar_dates를 결합하여 연도별 service_id별 실제 서비스 날짜를 만든다."""
    result = defaultdict(set)
    start = date(year, 1, 1)
    end = date(year, 12, 31)
    if not calendar.empty:
        for r in calendar.itertuples(index=False):
            sid = str(getattr(r, "service_id"))
            try:
                s = date(int(str(getattr(r, "start_date"))[:4]), int(str(getattr(r, "start_date"))[4:6]), int(str(getattr(r, "start_date"))[6:8]))
                e = date(int(str(getattr(r, "end_date"))[:4]), int(str(getattr(r, "end_date"))[4:6]), int(str(getattr(r, "end_date"))[6:8]))
            except Exception:
                continue
            s, e = max(s, start), min(e, end)
            if s > e:
                continue
            flags = {0: "monday", 1: "tuesday", 2: "wednesday", 3: "thursday", 4: "friday", 5: "saturday", 6: "sunday"}
            d = s
            while d <= e:
                if str(getattr(r, flags[d.weekday()], "0")) == "1":
                    result[sid].add(d)
                d += timedelta(days=1)
    if calendar_dates is not None and not calendar_dates.empty:
        require_columns(calendar_dates, ["service_id", "date", "exception_type"], "calendar_dates.txt")
        for r in calendar_dates.itertuples(index=False):
            try:
                ymd = str(getattr(r, "date"))
                d = date(int(ymd[:4]), int(ymd[4:6]), int(ymd[6:8]))
            except Exception:
                continue
            if d.year != year:
                continue
            sid = str(getattr(r, "service_id"))
            if str(getattr(r, "exception_type")) == "1":
                result[sid].add(d)
            elif str(getattr(r, "exception_type")) == "2":
                result[sid].discard(d)
    return result


def match_city(stop_name: str):
    """CONFIG에 명시된 도시명만 사용. 동명이인은 좌표/추측으로 분리하지 않는다."""
    s = norm(stop_name)
    hits = []
    for prov, cities in REGION_CITIES.items():
        for city in cities:
            if norm(city) and norm(city) in s and "서울" not in s:
                hits.append((prov, city))
    if len(hits) == 1:
        return hits[0][0], hits[0][1]
    if len(hits) > 1:
        return "", ""
    return "", ""


def destination_matches(stop_name: str, mode: str):
    s = norm(stop_name)
    found = []
    for dest, cfg in SEOUL_DESTINATIONS.items():
        if cfg["mode"] == "철도" and mode not in ("일반철도", "고속철도"):
            continue
        if cfg["mode"] == "시외·고속버스" and mode != "시외·고속버스":
            continue
        for p in cfg["patterns"]:
            if norm(p) and (norm(p) in s or s in norm(p)):
                found.append(dest)
                break
    return found


def pct(new, old):
    if pd.isna(old) or old == 0:
        return np.nan
    return (new - old) / old * 100.0

        require_columns(routes, ["route_id", "route_type"], "routes.txt")

        sdates = service_dates(calendar, calendar_dates, year)

            "평일": sum(d.weekday() < 5 for d in year_dates),

        for c in ["pickup_type", "drop_off_type"]:
            if c in header:

                일일운행횟수=("일일운행횟수", "sum"),
                평일평균운행횟수=("평일평균운행횟수", "sum"),
                토요일평균운행횟수=("토요일평균운행횟수", "sum"),
                일요일평균운행횟수=("일요일평균운행횟수", "sum"),
            ).reset_index()
            summary = base_grid.merge(agg, on=keys, how="left")
        else:
            summary = base_grid.copy()
        numeric_cols = ["노선수", "운행trip수", "연간운행횟수", "평일운행횟수", "토요일운행횟수", "일요일운행횟수", "일일운행횟수", "평일평균운행횟수", "토요일평균운행횟수", "일요일평균운행횟수"]
        for c in numeric_cols:
            if c not in summary:
                summary[c] = 0.0
            summary[c] = summary[c].fillna(0)
        summary["교통수단구분"] = np.where(summary["교통수단"].eq("시외·고속버스"), "버스", "철도")
        summary["분석연도"] = year
        # 사람이 읽는 요약표: 누적 연간 운행횟수는 제외하고 평균 운행횟수만 제공
        summary_view_cols = [
            "분석연도", "도", "출발시군", "서울도착지권역", "교통수단구분", "교통수단",
            "노선수", "일일운행횟수", "평일평균운행횟수", "토요일평균운행횟수", "일요일평균운행횟수"
        ]
        summary_view = summary[summary_view_cols].copy()

        # 시·군 합계: 도착지·교통수단별 합계와 전체 합계를 모두 제공
        total_keys = ["분석연도", "도", "출발시군", "교통수단", "교통수단구분"]
        total_values = ["노선수", "일일운행횟수", "평일평균운행횟수", "토요일평균운행횟수", "일요일평균운행횟수"]
        city_total = summary.groupby(total_keys, as_index=False)[total_values].sum()
        city_total["서울도착지권역"] = "전체 서울 목적지"
        city_total = city_total[["분석연도", "도", "출발시군", "서울도착지권역", "교통수단구분", "교통수단"] + total_values]
        # 버스/철도 구분을 주 분석축으로 쓸 수 있는 별도 요약행
        mode_total_keys = ["분석연도", "도", "출발시군", "교통수단구분"]
        mode_total = city_total.groupby(mode_total_keys, as_index=False)[total_values].sum()
        mode_total["서울도착지권역"] = "전체 서울 목적지"
        mode_total["교통수단"] = mode_total["교통수단구분"]
        mode_total = mode_total[city_total.columns]
        city_total = pd.concat([city_total, mode_total], ignore_index=True)

        # 노선별 패널
        if detail.empty:
            route_panel = pd.DataFrame(columns=["분석연도", "도", "출발시군", "서울도착지권역", "교통수단", "route_id", "노선명", "일일운행횟수"])
        else:
            route_panel = detail.groupby(["분석연도", "도", "출발시군", "서울도착지권역", "교통수단", "route_id", "route_short_name", "route_long_name"], as_index=False).agg(
                연간운행횟수=("연간운행횟수", "sum"), 평일운행횟수=("평일운행횟수", "sum"),
                토요일운행횟수=("토요일운행횟수", "sum"), 일요일운행횟수=("일요일운행횟수", "sum"),
                서비스일수_최대=("서비스일수", "max"), trip수=("trip_id", "nunique")
            )
            route_panel["일일운행횟수"] = route_panel["연간운행횟수"] / len(year_dates)
            route_panel["평일평균운행횟수"] = route_panel["평일운행횟수"] / weekday_denoms["평일"] if weekday_denoms["평일"] else np.nan
            route_panel["토요일평균운행횟수"] = route_panel["토요일운행횟수"] / weekday_denoms["토요일"] if weekday_denoms["토요일"] else np.nan
            route_panel["일요일평균운행횟수"] = route_panel["일요일운행횟수"] / weekday_denoms["일요일"] if weekday_denoms["일요일"] else np.nan
            route_panel["교통수단구분"] = np.where(route_panel["교통수단"].eq("시외·고속버스"), "버스", "철도")
            route_panel = route_panel.rename(columns={"route_short_name": "노선명", "route_long_name": "노선상세"})

        diagnostics_df = pd.DataFrame(diagnostics + suspicious)
        unmatched_df = pd.DataFrame(unmatched)
        if diagnostics_df.empty:
            diagnostics_df = pd.DataFrame(columns=["분석연도", "이상항목", "건수", "상세"])
        if unmatched_df.empty:
            unmatched_df = pd.DataFrame(columns=["분석연도", "trip_id", "출발지_stop_id", "출발지명", "사유"])

        # 연도별 파일
        output_dir.mkdir(parents=True, exist_ok=True)
        prefix = f"{year}_충청도_서울직행_"
        detail.to_csv(output_dir / f"{prefix}노선목록.csv", index=False, encoding="utf-8-sig")
        summary_view.to_csv(output_dir / f"{prefix}시군구_도착지별요약.csv", index=False, encoding="utf-8-sig")
        city_total.to_csv(output_dir / f"{prefix}시군별_서울행요약.csv", index=False, encoding="utf-8-sig")
        route_panel.to_csv(output_dir / f"{prefix}노선별요약.csv", index=False, encoding="utf-8-sig")
        diagnostics_df.to_csv(output_dir / f"{prefix}이상치진단.csv", index=False, encoding="utf-8-sig")
        unmatched_df.to_csv(output_dir / f"{prefix}출발지매칭실패.csv", index=False, encoding="utf-8-sig")
        route_type_diag.to_csv(output_dir / f"{prefix}route_type검증.csv", index=False, encoding="utf-8-sig")

        with pd.ExcelWriter(output_dir / f"{prefix}분석결과.xlsx", engine="openpyxl") as writer:
            detail.to_excel(writer, sheet_name="노선목록", index=False)
            summary_view.to_excel(writer, sheet_name="시군구_도착지별요약", index=False)
            city_total.to_excel(writer, sheet_name="시군별_서울행요약", index=False)
            route_panel.to_excel(writer, sheet_name="노선별요약", index=False)
            diagnostics_df.to_excel(writer, sheet_name="이상치진단", index=False)
            unmatched_df.to_excel(writer, sheet_name="출발지매칭실패", index=False)
            route_type_diag.to_excel(writer, sheet_name="route_type검증", index=False)

        return {"detail": detail, "summary": summary_view, "city_total": city_total, "route_panel": route_panel, "diagnostics": diagnostics_df, "unmatched": unmatched_df}
    finally:
        shutil.rmtree(work, ignore_errors=True)


def add_growth(df: pd.DataFrame, value_col: str, group_cols: list[str]):
    out = df.sort_values(group_cols + ["분석연도"]).copy()
    out["전년값"] = out.groupby(group_cols)[value_col].shift(1)
    out["전년대비증감"] = out[value_col] - out["전년값"]
    out["전년대비증감률(%)"] = np.where(out["전년값"].fillna(0).eq(0), np.nan, out["전년대비증감"] / out["전년값"] * 100)
    return out


def main():
    if not INPUT_PATHS:
        raise ValueError("INPUT_PATHS에 최소 한 개 연도의 GTFS 경로를 입력하십시오.")
    out = Path(OUTPUT_DIR)
    results = {}
    for year, src in sorted(INPUT_PATHS.items()):
        if not Path(src).exists():
            print(f"[{year}] 경로 없음, 건너뜀: {src}")
            continue
        print(f"[{year}] 분석 시작")
        results[year] = analyze_year(int(year), src, out)
        print(f"[{year}] 완료: 확정 상세행={len(results[year]['detail']):,}")

    if not results:
        raise RuntimeError("분석 가능한 연도가 없습니다.")

    all_summary = pd.concat([v["summary"] for v in results.values()], ignore_index=True)
    all_city_total = pd.concat([v["city_total"] for v in results.values()], ignore_index=True)
    all_route_panel = pd.concat([v["route_panel"] for v in results.values()], ignore_index=True)
    all_detail = pd.concat([v["detail"] for v in results.values()], ignore_index=True)
    all_diag = pd.concat([v["diagnostics"] for v in results.values()], ignore_index=True)
    all_unmatched = pd.concat([v["unmatched"] for v in results.values()], ignore_index=True)

    # 시군·도착지·교통수단별 전년 대비 노선수 및 일일운행횟수
    growth_group = ["도", "출발시군", "서울도착지권역", "교통수단"]
    panel = add_growth(all_summary, "노선수", growth_group)
    panel = panel.rename(columns={"전년값": "전년노선수", "전년대비증감": "노선수증감", "전년대비증감률(%)": "노선수증감률(%)"})
    tmp = add_growth(all_summary, "일일운행횟수", growth_group)
    panel["전년일일운행횟수"] = tmp["전년값"].values
    panel["일일운행횟수증감"] = tmp["전년대비증감"].values
    panel["일일운행횟수증감률(%)"] = tmp["전년대비증감률(%)"].values

    # 버스/철도별 요약과 요일별 변화율
    mode_group = ["도", "출발시군", "서울도착지권역", "교통수단구분"]
    mode_year = all_summary.groupby(["분석연도"] + mode_group, as_index=False)[["노선수", "일일운행횟수", "평일평균운행횟수", "토요일평균운행횟수", "일요일평균운행횟수"]].sum()
    mode_growth = add_growth(mode_year, "노선수", mode_group)
    mode_growth = mode_growth.rename(columns={"전년값": "전년노선수", "전년대비증감": "노선수증감", "전년대비증감률(%)": "노선수증감률(%)"})
    daily_growth = add_growth(mode_year, "일일운행횟수", mode_group)
    mode_growth["전년일일운행횟수"] = daily_growth["전년값"].values
    mode_growth["일일운행횟수증감"] = daily_growth["전년대비증감"].values
    mode_growth["일일운행횟수증감률(%)"] = daily_growth["전년대비증감률(%)"].values
    mode_growth = mode_growth[[
        "분석연도", "도", "출발시군", "서울도착지권역", "교통수단구분",
        "노선수", "일일운행횟수", "평일평균운행횟수", "토요일평균운행횟수", "일요일평균운행횟수",
        "전년노선수", "노선수증감", "노선수증감률(%)",
        "전년일일운행횟수", "일일운행횟수증감", "일일운행횟수증감률(%)"
    ]]
    # 노선별 유지·신규·소멸 및 노선별 증감률
    route_group = ["도", "출발시군", "서울도착지권역", "교통수단", "route_id"]
    route_growth = add_growth(all_route_panel, "일일운행횟수", route_group)
    route_growth["노선상태"] = np.select([
        route_growth["전년값"].isna() & route_growth["일일운행횟수"].gt(0),
        route_growth["전년값"].gt(0) & route_growth["일일운행횟수"].gt(0),
        route_growth["전년값"].gt(0) & route_growth["일일운행횟수"].eq(0),
    ], ["신규", "유지", "현재값0"], default="확인필요")
    route_growth["노선별일일운행횟수증감률(%)"] = np.where(route_growth["전년값"].fillna(0).eq(0), np.nan, route_growth["전년대비증감률(%)"])

    # 전년도에는 존재했지만 다음 연도에 완전히 사라진 노선도 별도로 복원
    years = sorted(results)
    disappeared = []
    for prev, curr in zip(years, years[1:]):
        a = set(map(tuple, all_route_panel[all_route_panel["분석연도"] == prev][route_group].to_records(index=False)))
        b = set(map(tuple, all_route_panel[all_route_panel["분석연도"] == curr][route_group].to_records(index=False)))
        for key in sorted(a - b):
            row = dict(zip(route_group, key))
            row.update({"분석연도": curr, "비교연도": prev, "노선상태": "폐지/미검출", "전년일일운행횟수": np.nan, "현재일일운행횟수": 0, "노선별일일운행횟수증감률(%)": -100.0})
            disappeared.append(row)
    disappeared_df = pd.DataFrame(disappeared)

    all_summary.to_csv(out / "충청도_시군구_도착지별요약.csv", index=False, encoding="utf-8-sig")
    panel.to_csv(out / "충청도_연도별증감_시군구별.csv", index=False, encoding="utf-8-sig")
    mode_growth.to_csv(out / "충청도_버스철도별증감.csv", index=False, encoding="utf-8-sig")
    route_growth.to_csv(out / "충청도_노선별증감.csv", index=False, encoding="utf-8-sig")
    disappeared_df.to_csv(out / "충청도_폐지또는미검출노선.csv", index=False, encoding="utf-8-sig")
    all_detail.to_csv(out / "충청도_노선목록.csv", index=False, encoding="utf-8-sig")
    all_diag.to_csv(out / "충청도_이상치진단.csv", index=False, encoding="utf-8-sig")
    all_unmatched.to_csv(out / "충청도_출발지매칭실패.csv", index=False, encoding="utf-8-sig")

    with pd.ExcelWriter(out / "충청도_GTFS_서울직행_통합분석결과.xlsx", engine="openpyxl") as writer:
        panel.to_excel(writer, sheet_name="연도별증감_시군구", index=False)
        mode_growth.to_excel(writer, sheet_name="버스철도별증감", index=False)
        route_growth.to_excel(writer, sheet_name="노선별증감", index=False)
        disappeared_df.to_excel(writer, sheet_name="폐지_미검출노선", index=False)
        all_summary.to_excel(writer, sheet_name="시군구_도착지별요약", index=False)
        all_city_total.to_excel(writer, sheet_name="시군별_서울행요약", index=False)
        all_detail.to_excel(writer, sheet_name="통합노선목록", index=False)
        all_diag.to_excel(writer, sheet_name="이상치진단", index=False)
        all_unmatched.to_excel(writer, sheet_name="출발지매칭실패", index=False)

    print("통합분석 완료")
    print("입력 연도:", sorted(results))
    print("통합 상세행:", len(all_detail))
    print("결과 폴더:", out)


if __name__ == "__main__":
    main()
